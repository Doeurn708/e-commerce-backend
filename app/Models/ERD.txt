        string password
        string role "customer, admin"
        timestamp created_at
    }

    CATEGORIES {
        bigint id PK
        string name
        string slug
        timestamp created_at
    }

    PRODUCTS {
        bigint id PK
        bigint category_id FK
        string name
        string slug
        text description
        decimal price
        integer stock
        string image
        timestamp created_at
    }

    ORDERS {
        bigint id PK
        bigint user_id FK
        decimal total_price
        string status "pending, processing, completed, cancelled"
        string shipping_address
        string phone
        timestamp created_at
    }

    ORDER_ITEMS {
        bigint id PK
        bigint order_id FK
        bigint product_id FK
        integer quantity
        decimal price "Price at time of purchase"
        timestamp created_at
    }

    ORDERS ||--|{ ORDER_ITEMS : "includes"

==== Example Flow ពេល User ទិញ Product =====

1. User Login
       ↓
2. Laravel verify account
       ↓
3. Sanctum return Token
       ↓
4. Vue save Token
       ↓
5. User GET /products
       ↓
6. Laravel GET products from DB
       ↓
7. User click Add To Cart
       ↓
8. Pinia save Cart
       ↓
9. User click Checkout
       ↓
10. Vue POST /orders
       ↓
11. Laravel validate Order
       ↓
12. Create orders
       ↓
13. Create order_items
       ↓
14. Reduce product quantity
       ↓
15. Database save
       ↓
16. Laravel return success
       ↓
17. Vue show Order Success
       ↓
18. User can see Order History
